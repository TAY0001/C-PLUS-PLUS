CREATE TABLE Shelter (
    Shelter_ID VARCHAR2(10) NOT NULL,
    Location VARCHAR2(15),
    PRIMARY KEY (Shelter_ID)
);

CREATE TABLE Animal (
    Animal_ID VARCHAR2(10) NOT NULL,
    Animal_Name VARCHAR2(15),
    Animal_Type VARCHAR2(15),
    Age NUMBER(3),
    Breed VARCHAR2(19),
    Year_Born NUMBER(4),
    Animal_Passport_ID VARCHAR2(15),
    Passport_ExpiryDate DATE,
    Shelter_ID VARCHAR2(10),
    PRIMARY KEY (Animal_ID),
    FOREIGN KEY (Shelter_ID) REFERENCES Shelter(Shelter_ID),
    CONSTRAINT chk_animal_age CHECK (Age >= 0),
    CONSTRAINT chk_year_born CHECK (Year_Born BETWEEN 2015 AND 2024) 
);

CREATE TABLE Customer (
    Customer_ID VARCHAR2(10) NOT NULL,
    Name VARCHAR2(30),
    Phone_Number VARCHAR2(12),
    Email VARCHAR2(26),
    Address VARCHAR2(50),
    Age NUMBER(2),
    Salary NUMBER(10,2),
    Trial_Program CHAR(3),
    Rental_Own VARCHAR2(10),
    Breeding_Experience VARCHAR2(25),
    PRIMARY KEY (Customer_ID),
    CONSTRAINT chk_age CHECK (Age >= 0),
    CONSTRAINT chk_salary CHECK (Salary >= 0),
    CONSTRAINT CHK_PHONE_NUMBER CHECK (Phone_Number LIKE '___-___-____'),
    CONSTRAINT CHK_EMAIL CHECK (Email LIKE '%@%.%')
);

CREATE TABLE Adoption (
    Adoption_ID VARCHAR2(11) NOT NULL,
    Adoption_Date DATE,
    Adoption_Fee NUMBER(9,2),
    Animal_ID VARCHAR2(10),
    Customer_ID VARCHAR2(10),
    PRIMARY KEY (Adoption_ID),
    FOREIGN KEY (Animal_ID) REFERENCES Animal(Animal_ID),
    FOREIGN KEY (Customer_ID) REFERENCES Customer(Customer_ID),
    CONSTRAINT chk_adoption_fee CHECK (Adoption_Fee >= 0)
);

CREATE TABLE Adoption_Counselor (
    Adoption_Counselor_ID VARCHAR2(21) NOT NULL,
    Counselor_Name VARCHAR2(25),
    PRIMARY KEY (Adoption_Counselor_ID)
);

CREATE TABLE Application (
    Application_ID VARCHAR2(15) NOT NULL,
    Submission_Date DATE,
    Accepted_or_Rejected VARCHAR2(20),
    Adoption_Counselor_ID VARCHAR2(21),
    Customer_ID VARCHAR2(10),
    PRIMARY KEY (Application_ID),
    FOREIGN KEY (Adoption_Counselor_ID) REFERENCES Adoption_Counselor(Adoption_Counselor_ID),
    FOREIGN KEY (Customer_ID) REFERENCES Customer(Customer_ID)
);

CREATE TABLE Trainer (
    Trainer_ID VARCHAR2(10) NOT NULL,
    Trainer_Name VARCHAR2(25),
    Phone_Number VARCHAR2(12),
    Email VARCHAR2(25),
    Experience_Level VARCHAR2(26),
    PRIMARY KEY (Trainer_ID),
    CONSTRAINT chk_trainer_phonenumber CHECK (Phone_Number LIKE '___-___-____'),
    CONSTRAINT chk_trainer_email CHECK (Email LIKE '%@%.%')
);

CREATE TABLE Training_Service (
    Training_Service_ID VARCHAR2(10) NOT NULL,
    Service_Name VARCHAR2(35),
    Customer_ID VARCHAR2(10),
    Trainer_ID VARCHAR2(10),
    PRIMARY KEY (Training_Service_ID),
    FOREIGN KEY (Customer_ID) REFERENCES Customer(Customer_ID),
    FOREIGN KEY (Trainer_ID) REFERENCES Trainer(Trainer_ID)
);

CREATE TABLE Training_Service_Details (
    Service_Name VARCHAR2(35) NOT NULL,
    Training_Service_Fee NUMBER(18,2),
    PRIMARY KEY (Service_Name),
    CONSTRAINT chk_training_service_fee CHECK (Training_Service_Fee >= 0)
);

CREATE TABLE Delivery_Services (
    Delivery_ID VARCHAR2(11) NOT NULL,
    Delivery_Method VARCHAR2(40),
    Delivery_Date DATE,
    Cost NUMBER(5,2),
    Customer_ID VARCHAR2(10),
    Adoption_ID VARCHAR2(11),
    PRIMARY KEY (Delivery_ID),
    FOREIGN KEY (Customer_ID) REFERENCES Customer(Customer_ID),
    FOREIGN KEY (Adoption_ID) REFERENCES Adoption(Adoption_ID),
    CONSTRAINT chk_delivery_cost CHECK (Cost >= 0)
);

CREATE TABLE Medical_Treatment (
    Medical_Treatment_ID VARCHAR2(20) NOT NULL,
    Medical_Name VARCHAR2(33),
    Medical_Treatment_Cost NUMBER(20,2),
    PRIMARY KEY (Medical_Treatment_ID),
    CONSTRAINT chk_treatment_cost CHECK (Medical_Treatment_Cost >= 0)
);

CREATE TABLE Animal_Treatment (
    Animal_Treatment_ID VARCHAR2(20) NOT NULL,
    Treatment_Date DATE,
    Medical_Treatment_ID VARCHAR2(20),
    Animal_ID VARCHAR2(10),
    PRIMARY KEY (Animal_Treatment_ID),
    FOREIGN KEY (Medical_Treatment_ID) REFERENCES Medical_Treatment(Medical_Treatment_ID),
    FOREIGN KEY (Animal_ID) REFERENCES Animal(Animal_ID)
);

CREATE TABLE Promotion (
    Promotion_Code VARCHAR2(15) NOT NULL,
    Start_Date DATE,
    PRIMARY KEY (Promotion_Code),
    CONSTRAINT chk_start_date CHECK (Start_Date IS NOT NULL)
);

CREATE TABLE Promotion_Details (
    Start_Date DATE NOT NULL,
    Expiration_Date DATE,
    PRIMARY KEY (Start_Date),
    CONSTRAINT chk_expiration_date CHECK (Expiration_Date >= Start_Date)
);

CREATE TABLE Payment (
    Payment_ID VARCHAR2(10) NOT NULL,
    Payment_Date DATE,
    Payment_Method VARCHAR2(20),
    Total_Price NUMBER(9,2),
    Payment_Amount NUMBER(12,2),
    Delivery_ID VARCHAR2(11),
    Service_Name VARCHAR2(35),
    Adoption_ID VARCHAR2(11),
    Promotion_Code VARCHAR2(15),
    PRIMARY KEY (Payment_ID),
    FOREIGN KEY (Adoption_ID) REFERENCES Adoption(Adoption_ID),
    FOREIGN KEY (Delivery_ID) REFERENCES Delivery_Services(Delivery_ID),
    FOREIGN KEY (Service_Name) REFERENCES Training_Service_Details(Service_Name),
    CONSTRAINT chk_total_price CHECK (Total_Price >= 0),
    CONSTRAINT chk_payment_amount CHECK (Payment_Amount >= 0)
);

CREATE TABLE Donation (
    Donation_ID VARCHAR2(10) NOT NULL,
    Donor_Name VARCHAR2(25),
    Donation_Amount NUMBER(18,2),
    Donation_Date DATE,
    Payment_Method VARCHAR2(20),
    Shelter_ID VARCHAR2(10),
    PRIMARY KEY (Donation_ID),
    FOREIGN KEY (Shelter_ID) REFERENCES Shelter(Shelter_ID),
    CONSTRAINT chk_donation_amount CHECK (Donation_Amount >= 0)
);
