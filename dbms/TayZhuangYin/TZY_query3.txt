ACCEPT Year1 PROMPT 'Enter the first year: '
ACCEPT Year2 PROMPT 'Enter the second year: '

SET PAGESIZE 35
SET LINESIZE 130

COLUMN cal_month_name FORMAT A17 HEADING 'Month'
COLUMN cal_quarter FORMAT A8 HEADING 'Quarter'
COLUMN lost_year1 FORMAT 999 HEADING 'Lost in &Year1'
COLUMN lost_year2 FORMAT 999 HEADING 'Lost in &Year2'
COLUMN lost_change FORMAT 999 HEADING 'Change'
COLUMN lost_pct_change FORMAT 999.99 HEADING 'Change(%)'
COLUMN damaged_year1 FORMAT 999 HEADING 'Damaged in &Year1'
COLUMN damaged_year2 FORMAT 999 HEADING 'Damaged in &Year2'
COLUMN damaged_change FORMAT 999 HEADING 'Change'
COLUMN damaged_pct_change FORMAT 999.99 HEADING 'Change(%)'

BREAK ON REPORT
COMPUTE SUM LABEL 'Grand Total (RM):' OF lost_year1 lost_year2 damaged_year1 damaged_year2 ON REPORT

TTITLE LEFT 'Date: ' _DATE SKIP 1 -
       CENTER 'Lost and Damaged Books Comparison &Year1 vs &Year2' SKIP 1 -
       RIGHT 'Page: ' SQL.PNO SKIP 2

CREATE OR REPLACE VIEW lost_damaged_comparison_view AS
WITH year_summary AS (
    SELECT
        d.cal_year,
        d.cal_month_name,
        d.cal_quarter,
        SUM(CASE WHEN f.fine_type = 'LOST_BOOK' THEN 1 ELSE 0 END) AS total_lost,
        SUM(CASE WHEN f.fine_type = 'DAMAGED' THEN 1 ELSE 0 END) AS total_damaged
    FROM fact_fines f
    JOIN dim_date d
        ON f.fine_date_key = d.date_key
    WHERE d.cal_year IN (&Year1, &Year2)
    GROUP BY d.cal_year, d.cal_quarter, d.cal_month_name
)
SELECT
    y1.cal_month_name,
    y1.cal_quarter,
    y1.total_lost AS lost_year1,
    y2.total_lost AS lost_year2,
    (y2.total_lost - y1.total_lost) AS lost_change,
    ROUND(DECODE(y1.total_lost,0,NULL,(y2.total_lost - y1.total_lost)/y1.total_lost*100),1) AS lost_pct_change,
    y1.total_damaged AS damaged_year1,
    y2.total_damaged AS damaged_year2,
    (y2.total_damaged - y1.total_damaged) AS damaged_change,
    ROUND(DECODE(y1.total_damaged,0,NULL,(y2.total_damaged - y1.total_damaged)/y1.total_damaged*100),1) AS damaged_pct_change
FROM year_summary y1
JOIN year_summary y2
    ON y1.cal_month_name = y2.cal_month_name
   AND y1.cal_year = &Year1
   AND y2.cal_year = &Year2
ORDER BY TO_DATE(y1.cal_month_name,'Month');

SELECT * FROM lost_damaged_comparison_view;
