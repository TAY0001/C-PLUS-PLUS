ACCEPT Start_Year PROMPT 'Enter the start year: '
ACCEPT End_Year PROMPT 'Enter the end year: '

SET PAGESIZE 35

COLUMN genre_name FORMAT A20 HEADING 'Genre Name'
COLUMN year FORMAT 9999 HEADING 'Year'
COLUMN holiday_revenue FORMAT 999,990.99 HEADING 'Holiday|Revenue(RM)'
COLUMN regular_revenue FORMAT 999,990.99 HEADING 'Regular|Revenue(RM)'
COLUMN weekend_revenue FORMAT 999,990.99 HEADING 'Weekend|Revenue(RM)'
COLUMN total_revenue FORMAT 9,999,990.99 HEADING 'Total|Revenue(RM)'
COLUMN yoy_growth FORMAT 990.0 HEADING 'YoY|Growth(%)'
COLUMN contribution FORMAT 990.00 HEADING 'Contribution|(%)'

BREAK ON genre_name SKIP 1 ON REPORT
COMPUTE SUM LABEL 'Genre Total: ' OF holiday_revenue regular_revenue weekend_revenue total_revenue ON genre_name

TTITLE LEFT 'Date: ' _DATE SKIP 1 - 
    CENTER '3-Year Book Sales Trend Analysis (&Start_Year - &End_Year)' SKIP 1 -

CREATE OR REPLACE VIEW three_year_sales_trend_view AS
WITH yearly_sales AS (
    SELECT 
        dd.cal_year,
        SUM(fbs.TotalAmount) AS total_yearly_sales
    FROM fact_book_sale fbs
    JOIN dim_date dd ON fbs.sale_date_key = dd.date_key
    WHERE dd.cal_year BETWEEN &Start_Year AND &End_Year
    GROUP BY dd.cal_year
),
yearly_genre_data AS (
    SELECT 
        dd.cal_year AS year,
        db.genre_name,
        SUM(CASE WHEN dd.holiday_ind = 'Y' THEN fbs.TotalAmount ELSE 0 END) AS holiday_revenue,
        SUM(CASE WHEN dd.holiday_ind = 'N' AND dd.weekend_ind = 'N' THEN fbs.TotalAmount ELSE 0 END) AS regular_revenue,
        SUM(CASE WHEN dd.weekend_ind = 'Y' AND dd.holiday_ind = 'N' THEN fbs.TotalAmount ELSE 0 END) AS weekend_revenue,
        SUM(fbs.TotalAmount) AS total_revenue
    FROM fact_book_sale fbs
    JOIN dim_book db ON fbs.book_key = db.book_key
    JOIN dim_date dd ON fbs.sale_date_key = dd.date_key
    WHERE dd.cal_year BETWEEN &Start_Year AND &End_Year
    GROUP BY dd.cal_year, db.genre_name
),
with_contribution AS (
    SELECT 
        ygd.year,
        ygd.genre_name,
        ygd.holiday_revenue,
        ygd.regular_revenue,
        ygd.weekend_revenue,
        ygd.total_revenue,
        ROUND((ygd.total_revenue / ys.total_yearly_sales) * 100, 2) AS contribution
    FROM yearly_genre_data ygd
    JOIN yearly_sales ys ON ygd.year = ys.cal_year
),
with_growth AS (
    SELECT 
        wc.*,
        ROUND(((wc.total_revenue - LAG(wc.total_revenue) 
               OVER (PARTITION BY wc.genre_name ORDER BY wc.year)) / 
               NULLIF(LAG(wc.total_revenue) 
               OVER (PARTITION BY wc.genre_name ORDER BY wc.year), 0)) * 100, 1) AS yoy_growth
    FROM with_contribution wc
)
SELECT 
    year,
    genre_name,
    holiday_revenue,
    regular_revenue,
    weekend_revenue,
    total_revenue,
    contribution,
    yoy_growth
FROM with_growth
ORDER BY genre_name, year;

SELECT * FROM three_year_sales_trend_view;
