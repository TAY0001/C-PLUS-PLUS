ACCEPT Start_Year PROMPT 'Enter the start year: '
ACCEPT End_Year PROMPT 'Enter the end year: '

SET PAGESIZE 35
SET LINESIZE 130
SET NEWPAGE 1

COLUMN cal_year FORMAT 9999 HEADING 'Year'
COLUMN cal_month_name FORMAT A14 HEADING 'Month'
COLUMN cal_quarter FORMAT A8 HEADING 'Quarter'
COLUMN total_fines FORMAT 9,999,999.99 HEADING 'Total Fines (RM)'
COLUMN collected_fines FORMAT 9,999,999.99 HEADING 'Collected Fines (RM)'
COLUMN outstanding_fines FORMAT 9,999,999.99 HEADING 'Outstanding Fines (RM)'
COLUMN recovery_rate FORMAT 990.00 HEADING 'Recovery|Rate (%)'
COLUMN unpaid_count FORMAT 999 HEADING 'Repeated Members with|Unpaid Cases'

BREAK ON cal_year SKIP PAGE ON REPORT
COMPUTE SUM LABEL 'Total (RM): ' OF total_fines collected_fines outstanding_fines unpaid_count ON cal_year

TTITLE LEFT 'Date: ' _DATE SKIP 1 - 
    CENTER 'Fine Recovery Efficiency Report for Years &Start_Year to &End_Year' SKIP 1 - 
    RIGHT 'Page: ' SQL.PNO SKIP 2

CREATE OR REPLACE VIEW fine_recovery_summary_view AS
WITH monthly_fines AS (
    SELECT 
        dd.cal_year,
        dd.cal_month_name,
        dd.cal_quarter,
        SUM(ff.fine_amount) AS total_fines,
        SUM(CASE WHEN ff.payment_status = 'PAID' THEN ff.fine_amount ELSE 0 END) AS collected_fines,
        SUM(CASE WHEN ff.payment_status = 'UNPAID' THEN ff.fine_amount ELSE 0 END) AS outstanding_fines
    FROM fact_fines ff
    JOIN dim_date dd ON ff.fine_date_key = dd.date_key
    WHERE dd.cal_year BETWEEN &Start_Year AND &End_Year
    GROUP BY dd.cal_year, dd.cal_month_name, dd.cal_quarter
),
unpaid_members AS (
    SELECT 
        dd.cal_year,
        dd.cal_month_name,
        dd.cal_quarter,
        fm.member_id,
        COUNT(*) AS unpaid_count
    FROM fact_fines ff
    JOIN dim_member fm ON ff.member_surrogate_key = fm.member_surrogate_key
    JOIN dim_date dd ON ff.fine_date_key = dd.date_key
    WHERE dd.cal_year BETWEEN &Start_Year AND &End_Year
      AND ff.payment_status = 'UNPAID'
    GROUP BY dd.cal_year, dd.cal_month_name, dd.cal_quarter, fm.member_id
    HAVING COUNT(*) >= 2
)
SELECT 
    mf.cal_year,
    mf.cal_month_name,
    mf.cal_quarter,
    mf.total_fines,
    mf.collected_fines,
    mf.outstanding_fines,
    ROUND((mf.collected_fines / NULLIF(mf.total_fines,0)) * 100, 2) AS recovery_rate,
    NVL(COUNT(DISTINCT um.member_id),0) AS unpaid_count
FROM monthly_fines mf
LEFT JOIN unpaid_members um
       ON mf.cal_year = um.cal_year
      AND mf.cal_month_name = um.cal_month_name
      AND mf.cal_quarter   = um.cal_quarter
GROUP BY mf.cal_year, mf.cal_month_name, mf.cal_quarter, 
         mf.total_fines, mf.collected_fines, mf.outstanding_fines
ORDER BY mf.cal_year, TO_DATE(mf.cal_month_name, 'MONTH');

SELECT * FROM fine_recovery_summary_view;
